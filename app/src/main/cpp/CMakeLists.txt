cmake_minimum_required(VERSION 3.22)
project(ultimateims_pairing LANGUAGES C CXX ASM)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${BORINGSSL_DIR}" "${CMAKE_BINARY_DIR}/boringssl" EXCLUDE_FROM_ALL)
set_target_properties(crypto PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
add_library(uims_pairing SHARED pairing.cpp)
target_compile_features(uims_pairing PRIVATE cxx_std_17)
target_include_directories(uims_pairing PRIVATE "${BORINGSSL_DIR}/src/include")
target_link_libraries(uims_pairing PRIVATE crypto)

if(ANDROID)
    target_link_options(uims_pairing PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_include_directories(uims_pairing PRIVATE ${JNI_INCLUDE_DIRS})
endif()
